use "git status" to see the status of the the file if something changed without commmit it 
use "git add " to stage the file 
use "git add -p " to stage with sepreting the commits
use "git commit -m " to commit with message 
use "git restore " to restore change from ur git 
use "git log" to see the history 
use "git --no-pager log -2 " to see ur last 2 commit history 
use 'git --no-pager log --since="5min" ' to see ur commits in the last 5mins 
use 'git log --format=" * %h %ad | %s %d [%an]" --date=short  ' to Show logs in a personalized format, including the commit hash, date, message, branch information, and author name
